// Command server runs the calculator API.
//
// Usage:
//
//        server              serve on $PORT (default 8081)
//        server healthcheck  exit 0 if the local server answers /healthz with 200
package main

import (
        "context"
        "errors"
        "fmt"
        "io"
        "log/slog"
        "net/http"
        "os"
        "os/signal"
        "syscall"
        "time"

        "fullstack-calculator/backend/internal/httpapi"
)

const defaultPort = "8081"

func main() {
        ctx, stop := signal.NotifyContext(context.Background(), os.Interrupt, syscall.SIGTERM)
        code := run(ctx, os.Args[1:], os.Getenv, os.Stdout, os.Stderr)
        stop()
        os.Exit(code)
}

// run is main without the process globals, so it can be tested. It returns
// the exit code.
func run(ctx context.Context, args []string, getenv func(string) string, stdout, stderr io.Writer) int {
        port := portFromEnv(getenv)

        if len(args) > 0 && args[0] == "healthcheck" {
                if err := checkHealth(ctx, "http://127.0.0.1:"+port+"/healthz"); err != nil {
                        fmt.Fprintln(stderr, err)
                        return 1
                }
                return 0
        }

        logger := slog.New(slog.NewJSONHandler(stdout, nil))
        srv := newServer(":"+port, httpapi.NewHandler(logger))
        logger.Info("listening", "addr", srv.Addr)
        if err := serve(ctx, srv, logger); err != nil {
                logger.Error("server stopped", "error", err)
                return 1
        }
        return 0
}

func portFromEnv(getenv func(string) string) string {
        if p := getenv("PORT"); p != "" {
                return p
        }
        return defaultPort
}

func newServer(addr string, h http.Handler) *http.Server {
        return &http.Server{
                Addr:              addr,
                Handler:           h,
                ReadHeaderTimeout: 5 * time.Second,
                ReadTimeout:       10 * time.Second,
                WriteTimeout:      10 * time.Second,
                IdleTimeout:       60 * time.Second,
        }
}

// checkHealth reports whether url answers with HTTP 200.
func checkHealth(ctx context.Context, url string) error {
        ctx, cancel := context.WithTimeout(ctx, 2*time.Second)
        defer cancel()
        req, err := http.NewRequestWithContext(ctx, http.MethodGet, url, nil)
        if err != nil {
                return err
        }
        resp, err := http.DefaultClient.Do(req)
        if err != nil {
                return err
        }
        defer resp.Body.Close()
        if resp.StatusCode != http.StatusOK {
                return fmt.Errorf("healthcheck: status %d", resp.StatusCode)
        }
        return nil
}

// serve runs srv until ctx is done, then shuts it down gracefully.
func serve(ctx context.Context, srv *http.Server, logger *slog.Logger) error {
        errCh := make(chan error, 1)
        go func() { errCh <- srv.ListenAndServe() }()

        select {
        case err := <-errCh:
                if errors.Is(err, http.ErrServerClosed) {
                        return nil
                }
                return err
        case <-ctx.Done():
        }

        logger.Info("shutting down")
        shutdownCtx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
        defer cancel()
        return srv.Shutdown(shutdownCtx)
}

// Package calculator implements decimal128 arithmetic over wire-format
// strings. It knows nothing about HTTP or JSON.
package calculator

import (
        "fmt"

        "github.com/cockroachdb/apd/v3"
)

// Operation identifies a supported calculator operation.
type Operation string

const (
        OpAdd      Operation = "add"
        OpSubtract Operation = "subtract"
        OpMultiply Operation = "multiply"
        OpDivide   Operation = "divide"
        OpPower    Operation = "power"
        OpSqrt     Operation = "sqrt"
        OpPercent  Operation = "percent"
)

// Result is a formatted result and whether it had to be rounded.
type Result struct {
        Value   string
        Rounded bool
}

// arity is the number of operands each operation takes.
var arity = map[Operation]int{
        OpAdd: 2, OpSubtract: 2, OpMultiply: 2, OpDivide: 2, OpPower: 2, OpSqrt: 1, OpPercent: 2,
}

// maxExponentMagnitude bounds power exponents before any work is done, so a
// request like 10^999999999 is rejected without touching the decimal engine.
const maxExponentMagnitude = 6000

var (
        errUnknownOperation = &Error{Code: CodeUnknownOperation,
                Message: "Operation must be one of: add, subtract, multiply, divide, power, sqrt, percent."}
        errDivisionByZero     = &Error{Code: CodeDivisionByZero, Message: "Cannot divide by zero."}
        errNegativeSquareRoot = &Error{Code: CodeNegativeSquareRoot, Message: "Cannot take the square root of a negative number."}
        errNonIntegerExponent = &Error{Code: CodeNonIntegerExponent, Message: "The exponent must be a whole number."}
        errExponentOutOfRange = &Error{Code: CodeResultOutOfRange,
                Message: fmt.Sprintf("The exponent must be between -%d and %d.", maxExponentMagnitude, maxExponentMagnitude)}
        errResultOutOfRange = &Error{Code: CodeResultOutOfRange, Message: "The result is too large or too small to represent."}

        maxExponent = apd.New(maxExponentMagnitude, 0)
        minExponent = apd.New(-maxExponentMagnitude, 0)
        hundred     = apd.New(100, 0)
)

// Calculate validates the operation and its operands, then computes the
// result in decimal128. Domain failures are returned as *Error.
func Calculate(op Operation, operands []string) (Result, error) {
        want, ok := arity[op]
        if !ok {
                return Result{}, errUnknownOperation
        }
        if len(operands) != want {
                return Result{}, operandCountError(op, want)
        }
        args := make([]*apd.Decimal, len(operands))
        for i, s := range operands {
                d, err := parseOperand(s, i+1)
                if err != nil {
                        return Result{}, err
                }
                args[i] = d
        }

        var d apd.Decimal
        cond, err := compute(op, &d, args)
        switch {
        case err != nil:
                return Result{}, err
        case cond.Overflow() || cond.Underflow():
                return Result{}, errResultOutOfRange
        case cond.DivisionByZero() || cond.DivisionUndefined():
                return Result{}, errDivisionByZero
        case cond.InvalidOperation():
                return Result{}, fmt.Errorf("calculator: invalid operation %s%v", op, operands)
        }
        return Result{Value: formatDecimal(&d), Rounded: cond.Inexact()}, nil
}

func compute(op Operation, d *apd.Decimal, args []*apd.Decimal) (apd.Condition, error) {
        switch op {
        case OpAdd:
                return decimal128.Add(d, args[0], args[1])
        case OpSubtract:
                return decimal128.Sub(d, args[0], args[1])
        case OpMultiply:
                return decimal128.Mul(d, args[0], args[1])
        case OpDivide:
                if args[1].IsZero() {
                        return 0, errDivisionByZero
                }
                return decimal128.Quo(d, args[0], args[1])
        case OpPower:
                return power(d, args[0], args[1])
        case OpSqrt:
                if args[0].Negative && !args[0].IsZero() {
                        return 0, errNegativeSquareRoot
                }
                return decimal128.Sqrt(d, args[0])
        case OpPercent:
                product, err := decimal128.Mul(d, args[0], args[1])
                if err != nil {
                        return product, err
                }
                quotient, err := decimal128.Quo(d, d, hundred)
                return product | quotient, err
        }
        return 0, fmt.Errorf("calculator: unhandled operation %q", op)
}

func power(d, base, exponent *apd.Decimal) (apd.Condition, error) {
        var frac apd.Decimal
        exponent.Modf(nil, &frac)
        if !frac.IsZero() {
                return 0, errNonIntegerExponent
        }
        if exponent.Cmp(maxExponent) > 0 || exponent.Cmp(minExponent) < 0 {
                return 0, errExponentOutOfRange
        }
        if exponent.IsZero() {
                d.SetInt64(1)
                return 0, nil
        }
        if base.IsZero() && exponent.Negative {
                return 0, errDivisionByZero
        }
        return decimal128.Pow(d, base, exponent)
}

func operandCountError(op Operation, want int) *Error {
        noun := "operands"
        if want == 1 {
                noun = "operand"
        }
        return &Error{Code: CodeInvalidOperandCount, Message: fmt.Sprintf("%s needs exactly %d %s.", op, want, noun)}
}

// Code is a stable, machine-readable identifier for a domain error.
type Code string

const (
        CodeUnknownOperation    Code = "UNKNOWN_OPERATION"
        CodeInvalidOperandCount Code = "INVALID_OPERAND_COUNT"
        CodeInvalidNumber       Code = "INVALID_NUMBER"
        CodeDivisionByZero      Code = "DIVISION_BY_ZERO"
        CodeNegativeSquareRoot  Code = "NEGATIVE_SQUARE_ROOT"
        CodeNonIntegerExponent  Code = "NON_INTEGER_EXPONENT"
        CodeResultOutOfRange    Code = "RESULT_OUT_OF_RANGE"
)

// Error is a domain error whose Message is safe to show to end users.
type Error struct {
        Code    Code
        Message string
}

func (e *Error) Error() string { return e.Message }

package calculator

import (
        "fmt"
        "regexp"
        "strings"

        "github.com/cockroachdb/apd/v3"
)

// decimal128 is the IEEE 754 decimal128 arithmetic context used for every
// operation. Traps are disabled: callers inspect the returned conditions.
// It is only ever read, so it is safe to share across goroutines.
var decimal128 = apd.Context{
        Precision:   34,
        MaxExponent: 6144,
        MinExponent: -6143,
        Rounding:    apd.RoundHalfEven,
        Traps:       0,
}

const maxOperandDigits = 34

// operandPattern is the strict wire grammar for operands: plain or scientific
// notation, no leading '+', no leading zeros, no bare decimal points.
var operandPattern = regexp.MustCompile(`^-?(0|[1-9][0-9]*)(?:\.([0-9]+))?(?:[eE][+-]?[0-9]+)?$`)

// parseOperand validates s against the wire grammar and the decimal128 limits.
// position is 1-based and appears in error messages.
func parseOperand(s string, position int) (*apd.Decimal, error) {
        m := operandPattern.FindStringSubmatch(s)
        if m == nil {
                return nil, invalidNumber(position, "is not a valid number")
        }
        if coefficient := strings.TrimLeft(m[1]+m[2], "0"); len(coefficient) > maxOperandDigits {
                return nil, invalidNumber(position, fmt.Sprintf("has more than %d digits", maxOperandDigits))
        }
        d, _, err := apd.NewFromString(s)
        if err != nil {
                return nil, invalidNumber(position, "is not a valid number")
        }
        cond, err := decimal128.Round(d, d)
        if err != nil || cond.Overflow() || cond.Underflow() || cond.Inexact() {
                return nil, invalidNumber(position, "is out of range")
        }
        return d, nil
}

// formatDecimal renders d for the wire. Plain notation is used while the
// adjusted exponent stays within [-6, 33]; scientific notation otherwise.
// Trailing fractional zeros are dropped and negative zero renders as "0".
func formatDecimal(d *apd.Decimal) string {
        var r apd.Decimal
        if _, _, err := decimal128.Reduce(&r, d); err != nil {
                r.Set(d)
        }
        if r.IsZero() {
                return "0"
        }
        adjusted := r.NumDigits() + int64(r.Exponent) - 1
        if adjusted >= -6 && adjusted <= 33 {
                return r.Text('f')
        }
        return r.Text('E')
}

func invalidNumber(position int, problem string) *Error {
        return &Error{Code: CodeInvalidNumber, Message: fmt.Sprintf("Operand %d %s.", position, problem)}
}

package httpapi

import (
        "bytes"
        "encoding/json"
        "errors"
        "io"
        "log/slog"
        "net/http"

        "fullstack-calculator/backend/internal/calculator"
)

const maxBodyBytes = 1024

type calculateRequest struct {
        Operation string    `json:"operation"`
        Operands  []*string `json:"operands"`
}

type calculateResponse struct {
        Result  string `json:"result"`
        Rounded bool   `json:"rounded"`
}

// NewHandler returns the API's root handler, wrapped with panic recovery and
// request logging.
func NewHandler(logger *slog.Logger) http.Handler {
        mux := http.NewServeMux()
        mux.HandleFunc("POST /api/v1/calculate", handleCalculate)
        mux.HandleFunc("/api/v1/calculate", methodNotAllowed("POST"))
        mux.HandleFunc("GET /healthz", func(w http.ResponseWriter, _ *http.Request) {
                writeJSON(w, http.StatusOK, map[string]string{"status": "ok"})
        })
        mux.HandleFunc("/healthz", methodNotAllowed("GET"))
        mux.HandleFunc("/", func(w http.ResponseWriter, _ *http.Request) {
                writeError(w, http.StatusNotFound, codeNotFound, "Not found.")
        })
        return withRecovery(logger, withRequestLog(logger, mux))
}

func handleCalculate(w http.ResponseWriter, r *http.Request) {
        body, err := io.ReadAll(http.MaxBytesReader(w, r.Body, maxBodyBytes))
        if err != nil {
                var tooLarge *http.MaxBytesError
                if errors.As(err, &tooLarge) {
                        writeError(w, http.StatusRequestEntityTooLarge, codePayloadTooLarge, "Request body must not exceed 1 KB.")
                        return
                }
                writeError(w, http.StatusBadRequest, codeInvalidJSON, invalidJSONMessage)
                return
        }

        req, ok := decodeCalculateRequest(body)
        if !ok {
                writeError(w, http.StatusBadRequest, codeInvalidJSON, invalidJSONMessage)
                return
        }

        // A null operand becomes "", which the domain reports as INVALID_NUMBER
        // with the operand's position.
        operands := make([]string, len(req.Operands))
        for i, o := range req.Operands {
                if o != nil {
                        operands[i] = *o
                }
        }

        res, err := calculator.Calculate(calculator.Operation(req.Operation), operands)
        if err != nil {
                var domain *calculator.Error
                if errors.As(err, &domain) {
                        writeError(w, statusFor(domain.Code), string(domain.Code), domain.Message)
                        return
                }
                writeError(w, http.StatusInternalServerError, codeInternalError, internalErrorMessage)
                return
        }
        writeJSON(w, http.StatusOK, calculateResponse{Result: res.Value, Rounded: res.Rounded})
}

const (
        invalidJSONMessage   = `Request body must be a JSON object with "operation" and "operands".`
        internalErrorMessage = "Internal server error."
)

// decodeCalculateRequest accepts exactly one JSON object with known fields.
func decodeCalculateRequest(body []byte) (calculateRequest, bool) {
        var req calculateRequest
        dec := json.NewDecoder(bytes.NewReader(body))
        dec.DisallowUnknownFields()
        if err := dec.Decode(&req); err != nil {
                return req, false
        }
        if err := dec.Decode(&struct{}{}); err != io.EOF {
                return req, false
        }
        return req, true
}

func methodNotAllowed(allow string) http.HandlerFunc {
        return func(w http.ResponseWriter, _ *http.Request) {
                w.Header().Set("Allow", allow)
                writeError(w, http.StatusMethodNotAllowed, codeMethodNotAllowed, "Method not allowed.")
        }
}

package httpapi

import (
        "log/slog"
        "net/http"
        "time"
)

// statusRecorder remembers the status code written through it.
type statusRecorder struct {
        http.ResponseWriter
        status int
}

func (r *statusRecorder) WriteHeader(status int) {
        r.status = status
        r.ResponseWriter.WriteHeader(status)
}

// withRequestLog logs one structured line per request.
func withRequestLog(logger *slog.Logger, next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                start := time.Now()
                rec := &statusRecorder{ResponseWriter: w, status: http.StatusOK}
                next.ServeHTTP(rec, r)
                logger.Info("request",
                        "method", r.Method,
                        "path", r.URL.Path,
                        "status", rec.status,
                        "duration", time.Since(start),
                )
        })
}

// withRecovery turns a panic into a JSON 500 so a bug never leaks a stack
// trace or an empty response to the client.
func withRecovery(logger *slog.Logger, next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                defer func() {
                        if v := recover(); v != nil {
                                logger.Error("panic serving request", "panic", v, "path", r.URL.Path)
                                writeError(w, http.StatusInternalServerError, codeInternalError, internalErrorMessage)
                        }
                }()
                next.ServeHTTP(w, r)
        })
}

package httpapi

import (
        "encoding/json"
        "net/http"

        "fullstack-calculator/backend/internal/calculator"
)

// HTTP-level error codes. Domain codes come from the calculator package.
const (
        codeInvalidJSON      = "INVALID_JSON"
        codePayloadTooLarge  = "PAYLOAD_TOO_LARGE"
        codeNotFound         = "NOT_FOUND"
        codeMethodNotAllowed = "METHOD_NOT_ALLOWED"
        codeInternalError    = "INTERNAL_ERROR"
)

type errorBody struct {
        Code    string `json:"code"`
        Message string `json:"message"`
}

type errorResponse struct {
        Error errorBody `json:"error"`
}

func writeJSON(w http.ResponseWriter, status int, v any) {
        w.Header().Set("Content-Type", "application/json")
        w.WriteHeader(status)
        _ = json.NewEncoder(w).Encode(v)
}

func writeError(w http.ResponseWriter, status int, code, message string) {
        writeJSON(w, status, errorResponse{Error: errorBody{Code: code, Message: message}})
}

// statusFor maps a domain error code to its HTTP status: 400 when the request
// is malformed, 422 when it is well-formed but the maths has no answer.
func statusFor(code calculator.Code) int {
        switch code {
        case calculator.CodeUnknownOperation, calculator.CodeInvalidOperandCount, calculator.CodeInvalidNumber:
                return http.StatusBadRequest
        case calculator.CodeDivisionByZero, calculator.CodeNegativeSquareRoot,
                calculator.CodeNonIntegerExponent, calculator.CodeResultOutOfRange:
                return http.StatusUnprocessableEntity
        }
        return http.StatusInternalServerError
}
